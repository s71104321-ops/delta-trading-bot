@app.route("/webhook", methods=["POST"])
def webhook():
    # Capture incoming JSON payload from TradingView
    incoming_data = request.json or {}
    print("Received Webhook Payload:", incoming_data)
    
    # Extract values sent by TradingView (using float for size to support 0.1 lots)
    product_id = int(incoming_data.get("product_id", 27))
    size = float(incoming_data.get("size", 0.1))
    side = str(incoming_data.get("side", "buy"))
    limit_price = str(incoming_data.get("limit_price", "50000"))
    
    try:
        # Initialize the authenticated Delta client for this request
        delta_client = DeltaRestClient(
            base_url=base_url,
            api_key=API_KEY,
            api_secret=API_SECRET,
            raise_for_status=False
        )
        
        # Execute the market/limit order on Delta Exchange India
        order_response = delta_client.place_order(
            product_id=product_id,
            size=size,
            side=side,
            limit_price=limit_price
        )
        
        print("Order Executed Successfully:", order_response)
        return jsonify(order_response), 200
        
    except Exception as e:
        print("Execution Error:", str(e))
        return jsonify({"success": False, "error": str(e)}), 500

if __name__ == "__main__":
    # Runs the local server on port 5000
    app.run(host="127.0.0.1", port=5000)
